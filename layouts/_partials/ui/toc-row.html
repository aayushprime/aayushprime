{{- /* §11.11 TOC row. Recursive — Hugo resolves recursion through partials, not through `define` blocks inside a partial. Do NOT wrap this in `with .h`: Hugo's tableofcontents.Heading.IsZero() is true whenever ID and Title are both empty, and a page whose headings start at h2 gets exactly such a synthetic root node holding the real headings as children. `with` would silently skip the whole subtree. Instead, a titleless node renders its children in place, at the same depth. dict: h (a Fragments heading), depth (int) */ -}} {{- $h := .h -}} {{- $depth := .depth -}} {{- if $h.Title -}} {{- if gt $depth 0 }} {{- /* §11.11: nested groups get a static guide line. */}} {{- end }} {{- $h.Title | plainify | htmlUnescape -}} {{- with $h.Headings }}
    {{- range . }} {{- partial "ui/toc-row.html" (dict "h" . "depth" (add $depth 1)) }} {{- end }}
{{- end }} {{- else -}} {{- /* Synthetic root: emit its children in place. */ -}} {{- range $h.Headings }} {{- partial "ui/toc-row.html" (dict "h" . "depth" $depth) }} {{- end -}} {{- end -}}
